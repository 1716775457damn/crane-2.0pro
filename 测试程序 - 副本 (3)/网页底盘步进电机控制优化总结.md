```javascript
// 连续转动控制
function startContinuousRotation(direction) {
    const motor = document.getElementById('motorSelect').value;
    const speed = document.getElementById('motorSpeed').value;
    const steps = direction === 'forward' ? 50 : -50;
    
    motorInterval = setInterval(() => {
        // 每200ms发送一次50步命令

            method: 'POST',
            body: `motor=${motor}&steps=${steps}&speed=${speed}`
        });
    }, 200);
}
```

## 🔧 技术实现细节

### 1. Web服务器端

#### 底盘控制API
```cpp
void handleControlChassis() {
    String direction = server.arg("direction");
    int speed = server.arg("speed").toInt();
    
    // 速度限制
    if (speed < 0) speed = 0;
    if (speed > 100) speed = 100;
    
    // 方向映射
    uint8_t chassisDirection = CHASSIS_STOP;
    if (direction == "forward") chassisDirection = CHASSIS_FORWARD;
    else if (direction == "backward") chassisDirection = CHASSIS_BACKWARD;
    else if (direction == "left") chassisDirection = CHASSIS_LEFT;
    else if (direction == "right") chassisDirection = CHASSIS_RIGHT;
    
    // 执行控制
    control_chassis_raw(chassisDirection, speed, 0);
}
```

#### 路由注册
```cpp
server.on("/control_chassis", HTTP_POST, handleControlChassis);
```

### 2. 前端JavaScript

#### 底盘连续控制
```javascript
let chassisInterval = null;

function startChassisMove(direction) {
    const speed = document.getElementById('chassisSpeed').value;
    
    chassisInterval = setInterval(() => {
        fetch('/control_chassis', {
            method: 'POST',
            body: `direction=${direction}&speed=${speed}`
        });
    }, 100); // 每100ms发送一次命令
}

function stopChassisMove() {
    if (chassisInterval) {
        clearInterval(chassisInterval);
        chassisInterval = null;
    }
    
    fetch('/control_chassis', {
        method: 'POST',
        body: 'direction=stop&speed=0'
    });
}
```

### 3. 串口命令解析

#### 支持速度参数的解析
```cpp
if (cmd.startsWith("chassis_to_x,")) {
    int firstComma = cmd.indexOf(',');
    int secondComma = cmd.indexOf(',', firstComma + 1);
    
    int target_x = cmd.substring(firstComma + 1, 
                   secondComma > 0 ? secondComma : cmd.length()).toInt();
    uint8_t speed = 0; // 默认速度
    
    if (secondComma > 0) {
        speed = cmd.substring(secondComma + 1).toInt();
    }
    
    chassis_move_to_x(target_x, speed);
}
```
